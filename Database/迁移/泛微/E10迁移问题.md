```
CREATE OR REPLACE PROCEDURE "eteams"."get_actural_table_count" () AS $$ begin
	declare v_sql_1 varchar(2000);
	declare scname varchar(63); 
	declare tbname varchar(63); 
	declare num int default 0;
	declare tbname_lists cursor for select a.table_schema,a.table_name from information_schema.tables a left join information_schema.views b on a.table_schema = b.table_schema and a.table_name = b.table_name 
	where a.table_schema not in('information_schema','mysql','performance_schema','sys') and b.table_schema is null;
	declare continue handler for not found set num=1;
	open tbname_lists; 
	fetch tbname_lists into scname,tbname; 
		while num <> 1 do
			set v_sql_1 = concat('insert into tmp_get_actural_table_count(table_name,table_cnt) select lower('',scname,'.',tbname,'') as table_name,count(*) as table_cnt from ',scname,'.`',tbname,'`');
			set @sql_1 = v_sql_1;
			prepare stmt1 from @sql_1;  
			execute stmt1;   
			deallocate prepare stmt1; 
			fetch tbname_lists into scname,tbname; 
		end while;
	close tbname_lists;
end;
```

## 正后可直接在 ksql 执行的版本（金仓 V8 / PostgreSQL 兼容）

```
CREATE OR REPLACE PROCEDURE "eteams"."get_actural_table_count"()
LANGUAGE plpgsql
AS $$
DECLARE
    v_sql   text;
    rec     record;
BEGIN
    FOR rec IN
        SELECT a.table_schema, a.table_name
        FROM information_schema.tables a
        WHERE a.table_schema NOT IN ('information_schema','pg_catalog','sys')
          AND a.table_type = 'BASE TABLE'
        ORDER BY a.table_schema, a.table_name
    LOOP
        v_sql := format(
            'INSERT INTO tmp_get_actural_table_count(table_name, table_cnt)
             SELECT lower(%L || ''.'' || %L), count(*)
             FROM %I.%I',
            rec.table_schema, rec.table_name,
            rec.table_schema, rec.table_name
        );
        EXECUTE v_sql;
    END LOOP;
END;
$$;
```
