
 #查看表的膨胀情况（更全面）
-------------------------------------------------------------------------------------------------------------
-- 使用扩展视图（如果已安装）
-- 需要先创建扩展
-- CREATE EXTENSION sys_stat_statements;

SELECT 
    schemaname,
    relname,
    n_tup_ins,
    n_tup_upd,
    n_tup_del,
    n_live_tup,
    n_dead_tup,
    ROUND((n_dead_tup::numeric * 100) / NULLIF((n_live_tup + n_dead_tup), 0), 2) AS dead_tuple_ratio,
    sys_size_pretty(sys_relation_size(schemaname || '.' || relname)) AS table_size,
    last_vacuum,
    last_autovacuum
FROM sys_stat_user_tables
ORDER BY dead_tuple_ratio DESC NULLS LAST;

 #使用系统表查询
-------------------------------------------------------------------------------
-- 查看所有表的死亡元组比例
SELECT 
    schemaname,
    relname,
    n_live_tup AS 活元组,
    n_dead_tup AS 死亡元组,
    CASE 
        WHEN (n_live_tup + n_dead_tup) > 0 
        THEN ROUND((n_dead_tup::numeric * 100) / (n_live_tup + n_dead_tup), 2)
        ELSE 0 
    END AS 死亡元组比例,
    last_vacuum,
    last_autovacuum,
    last_analyze,
    last_autoanalyze
FROM sys_stat_all_tables
WHERE (n_live_tup + n_dead_tup) > 0
ORDER BY 死亡元组比例 DESC;
