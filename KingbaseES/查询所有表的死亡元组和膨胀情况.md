-- 查询所有表的死亡元组和膨胀情况
SELECT
    schemaname,
    relname,
    n_live_tup AS "活元组数",
    n_dead_tup AS "死亡元组数",
    round(n_dead_tup::numeric * 100 / (n_live_tup + n_dead_tup), 2) AS "死亡元组占比(%)",
    last_vacuum,
    last_autovacuum,
    last_analyze,
    last_autoanalyze
FROM sys_stat_all_tables
WHERE (n_live_tup + n_dead_tup) > 0
ORDER BY n_dead_tup DESC;
